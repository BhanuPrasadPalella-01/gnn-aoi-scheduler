\section{Introduction}

% =================================================================== 2 PROBLEM
\begin{frame}{Problem Statement}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item A sink must track the readings of \hl{$M$ sensors} (temperature, traffic speed, \dots)
              over a shared wireless channel.
        \item Only \hl{$C \ll M$ sensors may transmit per slot}: the sink must choose \emph{which}.
        \item Classic schedulers minimise the \hl{Age of Information} (AoI) and poll whoever is stalest,
              ignoring \emph{content}: correlated neighbours are polled as if independent.
        \item This wastes capacity on \textbf{redundant data} and worsens the sink's estimate of the field.
      \end{itemize}
      \vspace{0.2em}

    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.66,every node/.style={transform shape}]
        \fill[accentlight!60,rounded corners=6pt] (-0.2,-0.2) rectangle (5.2,4.2);
        \foreach \x/\y/\c in {0.6/3.4/bad,1.2/3.0/bad,0.9/2.4/bad,3.6/3.5/accent,4.3/2.9/accent,2.4/1.2/good,3.1/0.6/good,1.8/0.5/good,4.4/1.0/accent}
          \node[circle,draw=\c,fill=\c!25,minimum size=0.42cm,inner sep=0] at (\x,\y) {};
        \draw[bad!70,thick] (0.6,3.4)--(1.2,3.0)--(0.9,2.4)--(0.6,3.4);
        \draw[good!70,thick] (2.4,1.2)--(3.1,0.6)--(1.8,0.5)--(2.4,1.2);
        \draw[accent!70,thick] (3.6,3.5)--(4.3,2.9);
        \node[lab,text width=2.2cm] at (2.55,3.15) {3 sensors, almost\\ the same value};
        \node[draw=ink,fill=white,rounded corners=3pt,font=\scriptsize\bfseries] (s) at (2.6,-1.1) {Sink: only $C$ polls / slot};
        \foreach \x/\y in {0.6/3.4,1.2/3.0,2.4/1.2}
          \draw[-{Stealth},dashed,muted] (\x,\y) to[bend right=8] (s.north);
      \end{tikzpicture}\\[0.4em]
      {\scriptsize\color{muted} Polling three correlated neighbours tells the sink little more than
      polling one of them.}

    \end{column}
  \end{columns}
  \vspace{0.2em}
      \begin{block}{Question}
        \small With only $C$ polls per slot, which sensors should transmit so that the sink's
        estimate of \emph{all} $M$ readings is as accurate as possible, \textbf{without letting any
        sensor go stale}?
      \end{block}
\end{frame}

% =================================================================== 3 OBJECTIVES
\begin{frame}{Objectives}
  \begin{enumerate}\setlength\itemsep{0.6em}
    \item \textbf{Model} the sink's knowledge with a Kalman filter on a correlation graph
          \emph{learned from data}, so uncertainty about a sensor drops when its neighbours report.
    \item \textbf{Design} a GNN scheduler (message passing over the correlation graph and the sink's
          posterior) that picks $C$ sensors per slot, trained end-to-end with \textbf{PPO} on the true
          estimation error.
    \item \textbf{Guarantee freshness}: no sensor's AoI may exceed a bound $F$, enforced by
          construction for every policy, not as a soft penalty.
    \item \textbf{Validate on real data}: 45-sensor Intel Berkeley Lab temperatures and 325-sensor
          PEMS-BAY freeway speeds, evaluated only on held-out test periods.
    \item \textbf{Compare} against AoI-based, correlation-heuristic and model-based (Kalman-greedy)
          schedulers with significance tests, plus an ablation of the message passing.
    \item \textbf{Demonstrate} the schedulers live on real recordings in a browser front end.
  \end{enumerate}
\end{frame}

% =================================================================== GAP
\begin{frame}{Research Gap and Novelty}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \sect{Research gap}
      \begin{itemize}\small\setlength\itemsep{0.5em}
        \item Correlation-aware AoI work assumes the correlation is \textbf{known in advance}; nobody learns it from data.
        \item Learned schedulers (deep RL, GNNs) optimise \textbf{freshness or throughput}, not how accurately the
              sink knows every sensor.
        \item Their graphs describe \textbf{radio interference}, not which sensors carry the same information.
        \item Freshness limits are \textbf{soft}: met only on average or with high probability.
        \item Very few results on \textbf{real multi-sensor recordings}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \sect{Novelty of this work}
      \begin{itemize}\small\setlength\itemsep{0.5em}
        \item[\color{good}\faCheck] Sensor \textbf{correlation graph learned from real data}.
        \item[\color{good}\faCheck] GNN looks at two graphs: who is correlated \emph{in general}, and who is redundant
              \emph{right now}.
        \item[\color{good}\faCheck] Scheduler \textbf{starts as the best classical method} (Kalman-greedy); RL only
              learns improvements.
        \item[\color{good}\faCheck] \textbf{Hard freshness guarantee} for every sensor.
        \item[\color{good}\faCheck] Tested on \textbf{real data}: 45 lab sensors and 325 freeway sensors, plus a live demo.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% =================================================================== SINK
\begin{frame}{Methodology (2/5): How the Sink Understands the Field}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \sect{Step 1: Learn each sensor's behaviour}
      \begin{itemize}\small
        \item From the training days only, learn how much each reading changes from one 5-minute slot to the next.
      \end{itemize}
      \sect{Step 2: Build the correlation graph}
      \begin{itemize}\small
        \item Connect two sensors if their \emph{unexpected} changes move together (correlation $\ge 0.3$).
        \item Example: two thermometers in the same room warm up together, so they are linked.
      \end{itemize}
      \sect{Step 3: Kalman filter at the sink}
      \begin{itemize}\small
        \item \textbf{Predict}: each slot, move every estimate forward; uncertainty grows.
        \item \textbf{Correct}: when readings arrive, fix the estimates; uncertainty shrinks.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Key idea}
        \small Polling one sensor also reduces the sink's uncertainty about its \textbf{correlated neighbours}.
        So some sensors are cheap to infer and need fewer polls.
      \end{block}
      \vspace{0.3em}
      \centering
      \begin{tikzpicture}[scale=0.9]
        \node[circle,draw=accent,fill=accent!40,minimum size=0.6cm,font=\scriptsize] (a) at (0,0) {A};
        \node[circle,draw=accent,fill=accent!15,minimum size=0.6cm,font=\scriptsize] (b) at (1.6,0.5) {B};
        \node[circle,draw=accent,fill=accent!15,minimum size=0.6cm,font=\scriptsize] (c) at (1.6,-0.6) {C};
        \node[circle,draw=muted,fill=white,minimum size=0.6cm,font=\scriptsize] (d) at (3.4,0) {D};
        \draw[accent,thick] (a)--(b) (a)--(c);
        \node[lab,text width=2.2cm] at (0,-1.2) {A transmits};
        \node[lab,text width=2.6cm] at (1.9,1.25) {B, C: sink now\\ fairly sure too};
        \node[lab,text width=2.2cm] at (3.4,-1.0) {D: not linked,\\ still unsure};
      \end{tikzpicture}\\[0.3em]
      {\scriptsize\color{muted} The \textbf{information gain} of a sensor = how much total uncertainty disappears if it
      transmits. Kalman-greedy and our GNN both use it.}
    \end{column}
  \end{columns}
\end{frame}

% =================================================================== GOAL + FRESHNESS
\begin{frame}{Methodology (3/5): Goal and Freshness Guarantee}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \sect{Goal}
      \begin{block}{}
        \small Keep the gap between the \textbf{true readings} and the \textbf{sink's estimates} as small as possible,
        averaged over all sensors and all time slots.
      \end{block}
      \sect{Rules in every slot}
      \begin{itemize}\small
        \item At most $C$ sensors transmit.
        \item No sensor stays silent for more than $F$ slots.
      \end{itemize}
      \sect{Age of Information (AoI)}
      {\small Slots since the sink last heard from a sensor. It resets to 1 when the sensor transmits:}\\[0.3em]
      \begin{tikzpicture}[x=0.62cm,y=0.5cm]
        \foreach \v [count=\i] in {1,2,3,4,1,2,3,1}
          \node[draw=rule,fill=paper,minimum size=0.5cm,font=\scriptsize] at (\i,0) {\v};
        \foreach \i in {5,8} \node[font=\tiny,color=good] at (\i,-0.9) {polled};
        \node[lab,anchor=east] at (0.5,0) {age};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \sect{Freshness guard (same for every scheduler)}
      \begin{enumerate}\small\setlength\itemsep{0.25em}
        \item If more sensors must report soon than the channel can carry in time, book the \textbf{most urgent} into
              this slot now (earliest deadline first).
        \item The scheduler fills only the remaining places.
        \item The simulator checks the rules every slot.
      \end{enumerate}
      \begin{block}{Our setting}
        \small $F$ = 3 $\times$ the round-robin cycle:\\
        Intel Lab: 22 slots (1.8 hours)\\
        PEMS-BAY: 24 slots (2 hours)
      \end{block}
      {\scriptsize\color{muted} A hard rule, not a penalty: the scheduler can never trade a sensor's freshness for
      lower error.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Methodology (4/5): GNN Scheduler Architecture}
  \centering
  \begin{tikzpicture}[node distance=0.45cm and 0.5cm]
    \node[st=muted] (in) {\textbf{Inputs}\\ 13 facts per sensor\\ + 2 graphs};
    \node[st=accent,right=of in] (emb) {\textbf{Encode}\\ each sensor as 64 numbers};
    \node[st=accent,right=of emb,text width=3.1cm] (mp) {\textbf{3 message-passing layers}\\ each sensor gathers info from its correlated neighbours};
    \node[st=accent,right=of mp] (h) {\textbf{Sensor summary}\\ state + neighbourhood};
    \node[st=ink,below=0.9cm of h,text width=4.1cm,xshift=-0.6cm] (logit) {\textbf{Score each sensor}\\ information gain + learned correction $-$ ``similar sensor already picked''};
    \node[st=good,left=of logit,text width=2.4cm] (mask) {\textbf{Block}\\ already picked or\\ already forced};
    \node[st=orangeb,left=of mask,text width=2.4cm] (pick) {\textbf{Pick}\\ the best-scoring\\ sensor};
    \node[st=ink,left=of pick,text width=2.4cm] (kal) {\textbf{Update}\\ sink's uncertainty\\ after this pick};

    \draw[arr] (in)--(emb); \draw[arr] (emb)--(mp); \draw[arr] (mp)--(h); \draw[arr] (h)--(logit);
    \draw[arr] (logit)--(mask); \draw[arr] (mask)--(pick); \draw[arr] (pick)--(kal);
    \draw[arr] (kal.north) -- ++(0,0.35) -| node[lab,pos=0.27,above]{repeat until the slot is full} (logit.north west);
  \end{tikzpicture}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}\scriptsize
        \item \textbf{13 facts per sensor}: its age, how unsure the sink is about it, how much it would teach the sink,
              how soon the freshness rule will force it, how fast it changes, how connected it is, \dots
        \item \textbf{Two graphs}: who is correlated in general, and who is redundant right now.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}\scriptsize
        \item Picks one sensor at a time, so the second pick knows about the first (no two near-identical sensors).
        \item Before training it behaves \textbf{exactly like Kalman-greedy}; training only adds improvements.
        \item Small model (137k parameters); one model works for 45 or 325 sensors.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% =================================================================== TRAINING
\begin{frame}{Methodology (5/5): Training with Reinforcement Learning (PPO)}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \centering
      \begin{tikzpicture}[node distance=0.32cm,
          stp/.style={draw=accent,fill=accent!8,rounded corners=3pt,text width=6.4cm,align=left,inner sep=5pt,font=\small}]
        \node[stp] (s1) {\textbf{1. Start} from Kalman-greedy (the best classical scheduler).};
        \node[stp,below=of s1] (s2) {\textbf{2. Practise}: run the scheduler on many networks: random simulated ones and
          real training days.};
        \node[stp,below=of s2] (s3) {\textbf{3. Reward}: lower estimation error at the sink $\Rightarrow$ higher reward.};
        \node[stp,below=of s3] (s4) {\textbf{4. Improve}: PPO makes choices that earned more reward slightly more likely,
          in small, safe steps.};
        \node[stp,below=of s4] (s5) {\textbf{5. Check}: every 10 rounds, test on validation data; keep the best version.};
        \foreach \a/\b in {s1/s2,s2/s3,s3/s4,s4/s5} \draw[arr] (\a)--(\b);
        \draw[arr] (s5.east) -- ++(0.35,0) |- node[lab,pos=0.25,right]{repeat\\ 400 rounds} (s2.east);
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.42\textwidth}
      \sect{Why these choices}
      \begin{itemize}\small\setlength\itemsep{0.45em}
        \item \textbf{Why RL?} Nobody knows the ``right'' sensors for each slot, so there are no labels to learn from.
              RL learns from the result: the error.
        \item \textbf{Why PPO?} Stable and widely used; its small update steps keep the strong starting point.
        \item \textbf{Real training days} teach it that the sink's model is never perfect.
        \item Test days are \textbf{never} seen in training.
        \item About 35 minutes on a laptop CPU.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion and Future Work}

% =================================================================== 23 CONCLUSION
\begin{frame}{Conclusion}
  \begin{itemize}\setlength\itemsep{0.7em}
    \item Built \textbf{GNN-AoI}: a scheduler that learns the sensors' correlation graph from data, tracks the field with a
          Kalman sink, and picks transmissions with a GNN pointer decoder trained by PPO.
    \item Every sensor stays fresh by construction (hard AoI bound), unlike probabilistic or penalty-based methods.
    \item On \textbf{real held-out data} it lowers the sink's error by \good{20\%} on 45 Intel Lab sensors and
          \good{2.4\%} on 325 PEMS-BAY sensors versus AoI-based scheduling, and beats Kalman-greedy in
          \textbf{every} test window on both datasets.
    \item On synthetic fields it halves the error of max-age scheduling and even beats greedy selection that
          knows the true covariance.
    \item Delivered as an open-source, tested package with reproducible experiments and a live browser demo of
          the transmissions.
  \end{itemize}
\end{frame}

% =================================================================== 24 FUTURE WORK
\begin{frame}{Future Work}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item \textbf{Lossy channels}: packet errors and retransmissions (HARQ, as in \cite{6}).
        \item \textbf{Online adaptation}: re-fit the correlation graph as the field drifts (seasons, new sensors).
        \item \textbf{Bigger networks}: train on $M>100$ and use sparse message passing for thousands of sensors.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item \textbf{Non-linear sinks}: replace the Kalman filter with a learned spatio-temporal predictor
              (e.g.\ graph-recurrent models \cite{24}).
        \item \textbf{Distributed scheduling}: each gateway decides locally with the same shared GNN.
        \item \textbf{Hardware pilot}: run the scheduler on a real LoRa / Zigbee testbed.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section*{Q\&A}
% =================================================================== EXPECTED QUESTIONS
\begin{frame}{Important Questions}
  \begin{itemize}\setlength\itemsep{0.6em}
    \item {\color{accent}\textbf{Why not simply poll the sensor that has waited longest?}}\\
      {\small Its neighbour may already tell the sink what it reads, so that poll is wasted. We poll the sensor that teaches the sink the most.}
    \item {\color{accent}\textbf{How do you make sure no sensor is ignored?}}\\
      {\small A hard freshness rule: any sensor silent for $F$ slots (about 2 hours) is polled first, for every scheduler.}
    \item {\color{accent}\textbf{Why reinforcement learning?}}\\
      {\small There is no ``correct answer'' for each slot to learn from; RL learns directly from the resulting error.}
    \item {\color{accent}\textbf{Why these two datasets?}}\\
      {\small Intel Lab is a real sensor network with strong correlation; PEMS-BAY is larger (325 sensors) and less redundant. Together they show where the idea helps a lot and where it helps a little.}
    \item {\color{accent}\textbf{Is the improvement real or luck?}}\\
      {\small Tested only on unseen days: ours beats Kalman-greedy on 10/10 (Intel) and 15/15 (PEMS-BAY) test windows (Wilcoxon test, $p<0.01$).}
  \end{itemize}
\end{frame}
